# === OMEGAFLOW MYCELIUM PIPELINE (GitHub Actions) ===
# Python as pure ETL: extract from scientific protocols, compute physics,
# output flat {"data":[...]} JSON for the Rust CDN. No UI, no plotting.

# Core
polars              # Rust-backed multi-threaded DataFrame (fast CSV/JSON filtering)
numpy               # Array backend for astropy/metpy

# Astronomy (ICRS / 4D spacetime)
astroquery          # Gold standard for VizieR, HEASARC, MAST, SIMBAD, Gaia TAP-Server
astropy             # VOTable parsing, TDB time, ICRS coordinates
skyfield            # Planetary/Satellite ephemerides (TLE/SPK propagation)

# Earth observation
earthaccess         # NASA Earthdata S3 access (MODIS, SWOT, GEDI, ICESat-2)
pystac-client       # STAC catalog search (AWS, Microsoft Planetary Computer)

# Meteorology & oceanography
xarray              # Multi-dimensional arrays (NetCDF, GRIB)
netCDF4             # xarray backend for NetCDF
erddapy             # Oceanographic data (NOAA buoys, NERACOOS, EMODnet)

# Geophysics & seismology
obspy               # FDSN earthquake catalogs & MiniSEED waveforms

# Biosphere & biodiversity
pygbif              # GBIF global species occurrences
pyinaturalist       # iNaturalist real-time observations
